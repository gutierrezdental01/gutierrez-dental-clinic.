Your current Apps Script doGet returns JSON. GitHub Pages cannot reliably fetch this cross-origin.
Replace ONLY your existing doGet(e) with the following read-only JSONP-capable version.
Keep CALENDAR_ID, CLINIC_TIMEZONE and getAvailableSlots unchanged.

function doGet(e) {
  const params = e && e.parameter ? e.parameter : {};
  const callback = params.callback;
  if (callback && callback !== 'gutierrezAvailabilityCallback') {
    return ContentService.createTextOutput('Invalid callback');
  }
  let response;
  try {
    if (!params.date || !params.procedure) throw new Error('Missing inputs');
    const slots = getAvailableSlots(params.date, params.procedure);
    response = { success: true, date: params.date, procedure: params.procedure, slots: slots };
  } catch (err) {
    response = { success: false, error: 'Availability unavailable' };
  }
  const json = JSON.stringify(response);
  if (callback) {
    return ContentService.createTextOutput('gutierrezAvailabilityCallback(' + json + ');')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}
